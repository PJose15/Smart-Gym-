```typescript
import { z } from 'zod';
export const machineCreateSchema = z.object({
  name: z.string().min(2).max(80).trim(),
  target_muscles: z.array(z.string().min(1).max(40)).min(1).max(10),
  movement_pattern: z.enum(['push','pull','squat','hinge','carry','core','isolation','unknown']).optional().default('unknown'),
  equipment_type: z.enum(['machine','cable','dumbbell','barbell','bodyweight','smith','cardio','unknown']).optional().default('machine'),
  difficulty: z.enum(['beginner','intermediate','advanced']).optional().default('beginner'),
  setup_steps: z.array(z.string().max(200)).max(15).optional().default([]),
  safety_cues: z.array(z.string().max(200)).max(15).optional().default([]),
});
export type MachineCreateInput = z.infer<typeof machineCreateSchema>;
```

Create `apps/web-admin/src/app/api/machines/route.ts` POST handler following existing API route conventions:
1. `verifyStaff('owner')` — return early if NextResponse
2. `checkRateLimit(\`machines-create:${user_id}\`, 20, 60_000)`
3. Parse + validate body with machineCreateSchema (400 + `parsed.error.flatten().fieldErrors` on failure)
4. Fetch `gyms.slug` via `admin.from('gyms').select('slug').eq('id', gym_id).single()`
5. `const qr_slug = generateQrSlug(gym.slug, name)`
6. Generate `common_mistakes` server-side: `try { mistakes = await generateMachineMistakes({ machineName: name, targetMuscles: target_muscles, setupSteps: setup_steps }) } catch { mistakes = [] }` (use the pure `@nexera/ai-assist` function only — do NOT call the edge-function client from a server route)
7. Insert with explicit columns matching the page's payload: `{ gym_id, name, qr_slug, target_muscles, setup_steps, safety_cues, common_mistakes, cue_version: 1, cue_source: mistakes.length > 0 ? 8. On insert error code '23505' (qr_slug collision): retry once with `qr_slug + '-' + 4-char random suffix`
9. Return 201 `NextResponse.json({ id, name, qr_slug }, { status: 201 })` — `.select('id, name, qr_slug').single()` on insert
